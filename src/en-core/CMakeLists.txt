# en-core: the engine's core (ARCH §2.2). Its v0.1 slice (the first in-game build, part 5): the lifecycle without the
# store (ARCH §9), the offline and multiplayer gate with its hysteresis (ARCH §5.1), and the ladder's L1 and L2 with
# Dormant (ARCH §6.8). Pure: no I/O, no threads, no clocks, no Win32 (ARCH §2.3). The DLL links it (en_dll_module).
#
# Augur runs in every step (ARCH §8): en-core links en-augur, and builds in the shipped rule files
# (content/eldenring/augur/*.toml), read here at configure time into builtin_rules.cpp, since v0.1 loads no content.
file(GLOB EN_CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
file(GLOB EN_AUGUR_RULE_FILES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/content/eldenring/augur/*.toml)
list(SORT EN_AUGUR_RULE_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${EN_AUGUR_RULE_FILES})
set(_en_builtin ${CMAKE_CURRENT_BINARY_DIR}/builtin_rules.cpp)
set(_en_body "// Generated by src/en-core/CMakeLists.txt from content/eldenring/augur/*.toml: do not edit.\n")
string(APPEND _en_body "#include \"en/core/core.hpp\"\n\n#include <array>\n\nnamespace en::core {\nnamespace {\n")
set(_en_entries "")
set(_en_n 0)
foreach(_en_file IN LISTS EN_AUGUR_RULE_FILES)
  file(READ ${_en_file} _en_text)
  if(_en_text MATCHES "\\)EN_RULES\"")
    message(FATAL_ERROR "${_en_file} holds the raw string's delimiter")
  endif()
  get_filename_component(_en_name ${_en_file} NAME)
  string(APPEND _en_body "constexpr std::string_view kFile${_en_n} = R\"EN_RULES(${_en_text})EN_RULES\";\n")
  string(APPEND _en_entries "    BuiltinFile{\"${_en_name}\", kFile${_en_n}},\n")
  math(EXPR _en_n "${_en_n} + 1")
endforeach()
string(APPEND _en_body "constexpr std::array<BuiltinFile, ${_en_n}> kFiles = {\n${_en_entries}};\n}  // namespace\n\n")
string(APPEND _en_body "std::span<const BuiltinFile> builtin_rule_files() noexcept { return kFiles; }\n\n}  // namespace en::core\n")
file(WRITE ${_en_builtin}.in "${_en_body}")
configure_file(${_en_builtin}.in ${_en_builtin} COPYONLY)  # rewritten only when it changes
add_library(en_core STATIC ${EN_CORE_SOURCES} ${_en_builtin})
en_dll_module(en_core)
target_include_directories(en_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(en_core PUBLIC en_proto en_augur)

if(EN_BUILD_TESTS)
  file(GLOB EN_CORE_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp)
  add_executable(en_core_tests ${EN_CORE_TESTS})
  target_link_libraries(en_core_tests PRIVATE en_core en_doctest)
  target_compile_definitions(en_core_tests PRIVATE EN_AUGUR_CONTENT="${PROJECT_SOURCE_DIR}/content/eldenring/augur")
  add_test(NAME en_core_tests COMMAND en_core_tests)
endif()
